#3
tail -n +2 ../data/fasta/e_coli.fasta | tr -d '\r\n' | wc -c
# Answer 4686137
# tail excludes the FASTA header, tr -d '\r\n' removes carriage returns and newlines, and wc -c counts the remaining bytes. As each nucleotide symbol is a single ASCII character, the result (4686137) represents the total number of sequence positions, including ambiguous bases such as N, R, Y, M and W.

#4
tail -n +2 ../data/fasta/e_coli.fasta | tr -d '\r\n' | tr '[:lower:]' '[:upper:]' | grep -o 'ATGC' | wc -l
# Answer 21968
# tail removes the header, tr joins wrapped sequence lines and converts bases to uppercase, and grep -o outputs each literal ATGC match separately. wc -l counts these matches (21968). Joining lines ensures motifs spanning line breaks are detected, while retaining ambiguous bases prevents artificial matches.

tail -n +2 ../data/fasta/e_coli.fasta | tr -d '\r\n' | awk '{seq=toupper($0); s1=seq; s2=seq; AT=gsub(/[AT]/,"",s1); GC=gsub(/[GC]/,"",s2); print AT/GC}'
# Answer 0.969228
# tail excludes the FASTA header, and tr joins the sequence lines. awk converts the sequence to uppercase, counts A/T and G/C separately using gsub, and calculates their decimal ratio (A+T)/(G+C). The result (0.969228) represents approximately 0.969 AT bases per GC base. Ambiguous bases are excluded because their identities cannot be determined.
